\section{Even and Odd Numbers}

\textbf{Definition.} An integer $n$ is \emph{even} if $n = 2k$ for some
integer $k$, and \emph{odd} if $n = 2k + 1$ for some integer $k$.

\textbf{Theorem.} The sum of two even integers is even.

\textbf{Proof.} Let $m = 2k$ and $n = 2l$. Then
\[
  m + n = 2k + 2l = 2(k + l).
\]
Since $k + l$ is an integer, $m + n$ is even.

\textbf{Theorem.} The product of two odd integers is odd.

\textbf{Remark.} The sum of two odd integers is always even.
